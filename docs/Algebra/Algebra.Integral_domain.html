  Lemma nonzero_product_iff_nonzero_factors :
      ∀ x y : T,
      ¬eq (mul x y) zero ↔ ¬eq x zero ∧ ¬eq y zero.
  Proof.
    setoid_rewrite ring_zero_product_iff_zero_factor.
    intuition.
  Qed.
